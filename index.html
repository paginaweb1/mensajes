async function compartir(btn, index){ let card = btn.closest(".card"); let titulo = card.querySelector(".titulo"); /* ocultar titulo */ titulo.style.display = "none"; /* crear contenedor tipo flyer */ let flyer = document.createElement("div"); flyer.style.width = "600px"; flyer.style.padding = "20px"; flyer.style.background = "#0b1c2c"; flyer.style.display = "flex"; flyer.style.flexDirection = "column"; flyer.style.alignItems = "center"; flyer.style.borderRadius = "20px"; /* LOGO */ let logo = document.createElement("img"); logo.src = "logocasaroca.png"; logo.style.width = "120px"; logo.style.marginBottom = "15px"; /* imagen del post */ let imagen = card.querySelector("img").cloneNode(); imagen.style.width = "100%"; imagen.style.borderRadius = "15px"; /* texto abajo */ let texto = document.createElement("p"); texto.innerText = "Te invito a la iglesia Casa Roca 🙌"; texto.style.color = "#f5d27a"; texto.style.marginTop = "15px"; texto.style.textAlign = "center"; texto.style.fontWeight = "bold"; /* armar flyer */ flyer.appendChild(logo); flyer.appendChild(imagen); flyer.appendChild(texto); /* agregar temporal al DOM */ document.body.appendChild(flyer); /* capturar */ const canvas = await html2canvas(flyer, { backgroundColor: null, scale: 2, useCORS: true }); /* eliminar flyer del DOM */ document.body.removeChild(flyer); /* restaurar titulo */ titulo.style.display = "block"; canvas.toBlob(blob => { const file = new File([blob], `post_${index}.png`, { type: "image/png" }); if (navigator.canShare && navigator.canShare({ files: [file] })) { navigator.share({ files: [file], title: "La Silla Vacía", text: "Te invito a la iglesia Casa Roca 🙌" }); } else { let link = document.createElement("a"); link.href = canvas.toDataURL(); link.download = `post_${index}.png`; link.click(); } }); }
